cmake_minimum_required(VERSION 3.0.2)
project(fr3_allegro_sim)
find_package(catkin REQUIRED COMPONENTS gazebo_dev gazebo_ros_control pluginlib)
catkin_package()
add_library(fr3_allegro_hw_sim SHARED src/bounded_revolute_hw_sim.cpp)
target_compile_features(fr3_allegro_hw_sim PRIVATE cxx_std_14)
target_include_directories(fr3_allegro_hw_sim PRIVATE ${catkin_INCLUDE_DIRS})
target_link_libraries(fr3_allegro_hw_sim ${catkin_LIBRARIES})
install(TARGETS fr3_allegro_hw_sim LIBRARY DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION})
catkin_install_python(PROGRAMS scripts/motion_demo.py DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(DIRECTORY config launch meshes urdf worlds DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
install(FILES model.config robot_hw_sim_plugins.xml DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
